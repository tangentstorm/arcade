function loseLife() {
    soundManager.play('fall');
    if (ballsLeft == 0) {
        // game over
        //posted_scores = false;
    } else {
        if (el = $("spare" + ballsLeft)) {
	    el.style.display = 'none';
	}
        ballsLeft--;
        ball.stickToPaddle();
    }
}
var ballsLeft;
var SPARES_START = 3;

function level_init(level) {
    bricks.each(function (b) { b.reset() });
    Brick.count = bricks.length;
    current_level = level;
    paddle.center();
    ball.stickToPaddle();
    Element.update($('level'), current_level.toString());
}
